# Training-phase dependencies. NOT installed by requirements.txt, which is the
# data-prep + orchestrator environment and is deliberately CPU-only so it
# installs on a laptop.
#
# Install on the GPU box only:
#     pip install -r requirements-training.txt
#
# Versions are the ones pinned in notebooks/kaggle_t4_validation.py. Keep the
# two in sync — a version skew between the Kaggle validation run and the RunPod
# training run means the thing that was validated is not the thing that runs.

torch>=2.4                # provided by the Kaggle/RunPod image; listed for clarity
transformers==4.57.1      # DeepSeek-R1-Distill chat template support.
                          # Older versions drop the <think> format silently,
                          # which is the exact failure the validation run tests.
peft==0.18.0              # LoRA / QLoRA adapters
bitsandbytes==0.48.2      # 4-bit NF4 quantization + paged_adamw_8bit optimizer
accelerate==1.11.0        # device placement; used internally by Trainer
datasets==4.8.5           # matches requirements.txt so tokenisation is identical
pyyaml>=6                 # config loading

# Optional, not installed by default:
#   unsloth   ~30% less memory and faster, but its install frequently conflicts
#             with Kaggle's pinned torch build. train_lora.py auto-detects it
#             and falls back to transformers+PEFT, which is the path validated
#             on Kaggle and the one that runs on RunPod.
#   wandb     set WANDB_API_KEY to enable; train_lora.py honours WANDB_DISABLED
#             and WANDB_MODE, and reports to stdout only when absent.
